## Download files from NCBI

# generate list of SRR numbers
esearch -db sra -query "PRJNA1008718" | efetch -format runinfo | cut -d ',' -f 1 | tail -n +2 > bonasa_SRR.numbers

# prefetch
prefetch -O ./output_directory --option-file SRR.numbers   
prefetch -0 $OUTDIR --option-file bonasa_SRR.numbers

# validate (for loop) with vdb-validate
OUTDIR="/home/las80898/bonasa/reads"
cd $OUTDIR

for SRR in $OUTDIR
do
vdb-validate SRR*
done

# Extract files with fastq-dump
OUTDIR="/home/las80898/bonasa/reads"
cd $OUTDIR

# cycle through directories and use vdb-validate
for SRR in $OUTDIR
do
fastq-dump --split-3 SRR*
done

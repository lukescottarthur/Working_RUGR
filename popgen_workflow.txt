This is a workflow for bonasa popgen analyses
This begins with vcf.gz files that have been filtered.

~~For all analyses:

1. Merging vcf files into cohort
# Merge per-sample vcf.gz files into a cohort VCF
bcftools merge -O z -o cohort.vcf.gz *.vcf.gz
bcftools index -t cohort.vcf.gz

# Keep only biallelic SNPs
bcftools view -O z -o biallelic_snps.vcf.gz \
  -m2 -M2 -v snps biallelic_snps.vcf.gz

2. relatedness with Plink2 -make king table (instead of KING)
plink2 --vcf input.vcf.gz --make-king-table --out results   
# add code to remove pairs

3a. Split into variant and invariant sites (MAF filters remove invariant sites)
#extract invariant
vcftools --gzvcf input.vcf.gz \
  --max-maf 0 \
  --recode --stdout | bgzip -c > invariant.vcf.gz   

3b. MAF with plink
# MAF filter + genotype rate filter
plink2 --bfile cohort \
  --maf 0.05 \
  --geno 0.05 \
  --hwe 1e-6 \
  --make-bed \
  --out cohort_maf_filtered \
  --allow-extra-chr

3c. index and concatenate invariant and variant files

tabix invariant.vcf.gz
tabix variants.vcf.gz

bcftools concat \
  --allow-overlaps \
  variants.vcf.gz invariant.vcf.gz \
  -O z -o combined_filtered.vcf.gz   

4. Save two datasets, one for structure, ROH, PCA; the other for the rest

~~~STRUCTURE, ROH, PCA:

popgen_A_1

1. LD pruning
# LD pruning (window=50kb, step=10 SNPs, r²<0.1 for PCA/ADMIXTURE)
# Use r²<0.3–0.5 for GWAS-adjacent analyses
plink2 --bfile cohort_maf_filtered \
  --indep-pairwise 50 10 0.1 \
  --out pruned_snps \
  --allow-extra-chr

plink2 --bfile cohort_maf_filtered \
  --extract pruned_snps.prune.in \
  --make-bed \
  --out cohort_LD_pruned \
  --allow-extra-chr

2. Population structure PCA with plink
# PLINK2 PCA (use LD-pruned dataset)
plink2 --bfile cohort_LD_pruned \
  --pca 20 \
  --out cohort_pca \
  --allow-extra-chr

3. admixture
# Run for K=1 to K=10, use cross-validation to select best K
for K in $(seq 1 10); do
  admixture --cv cohort_LD_pruned.bed $K | tee log_K${K}.out
done

# Extract CV errors to select optimal K
grep "CV error" log_K*.out | sort -t: -k2 -n

4. inbreeding and ROH
# BCFtools ROH (HMM-based, best for WGS)
bcftools roh --AF-tag AF -O r -o roh_output.txt filtered.vcf.gz

#Calculate F_ROH as: sum of ROH length (>1Mb) / total autosomal genome length
